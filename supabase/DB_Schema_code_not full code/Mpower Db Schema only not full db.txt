  subissue_id uuid NOT NULL,
  language text NOT NULL,
  label text NOT NULL,
  CONSTRAINT complaint_subissue_translations_pkey PRIMARY KEY (id),
  CONSTRAINT complaint_subissue_translations_subissue_id_fkey FOREIGN KEY (subissue_id) REFERENCES public.complaint_subissues(id)
);
CREATE TABLE public.complaint_attachments (
  id uuid NOT NULL DEFAULT gen_random_uuid(),
  complaint_id uuid NOT NULL,
  file_type text NOT NULL CHECK (file_type = ANY (ARRAY['photo'::text, 'video'::text])),
  storage_path text NOT NULL,
  file_size_bytes bigint,
  caption text,
  uploaded_by_citizen_id uuid,
  uploaded_by_user_id uuid,
  created_at timestamp with time zone NOT NULL DEFAULT now(),
  CONSTRAINT complaint_attachments_pkey PRIMARY KEY (id),
  CONSTRAINT complaint_attachments_complaint_id_fkey FOREIGN KEY (complaint_id) REFERENCES public.complaints(id),
  CONSTRAINT complaint_attachments_uploaded_by_citizen_id_fkey FOREIGN KEY (uploaded_by_citizen_id) REFERENCES public.citizens(id),
  CONSTRAINT complaint_attachments_uploaded_by_user_id_fkey FOREIGN KEY (uploaded_by_user_id) REFERENCES public.users(id)
);
CREATE TABLE public.ui_labels (
  id uuid NOT NULL DEFAULT gen_random_uuid(),
  app_id uuid,
  label_key text NOT NULL,
  label_en text NOT NULL,
  CONSTRAINT ui_labels_pkey PRIMARY KEY (id),
  CONSTRAINT ui_labels_app_id_fkey FOREIGN KEY (app_id) REFERENCES public.apps(id)
);
CREATE TABLE public.ui_label_translations (
  id uuid NOT NULL DEFAULT gen_random_uuid(),
  label_id uuid NOT NULL,
  language text NOT NULL,
  label text NOT NULL,
  CONSTRAINT ui_label_translations_pkey PRIMARY KEY (id),
  CONSTRAINT ui_label_translations_label_id_fkey FOREIGN KEY (label_id) REFERENCES public.ui_labels(id)
);
CREATE TABLE public.staff_verification_requests (
  id uuid NOT NULL DEFAULT gen_random_uuid(),
  app_id uuid NOT NULL,
  requested_role text NOT NULL CHECK (requested_role = ANY (ARRAY['representative'::text, 'authority'::text, 'grievance_admin'::text])),
  full_name text NOT NULL,
  phone text NOT NULL,
  claimed_constituency_id uuid,
  claimed_authority_title text,
  verification_method text CHECK (verification_method = ANY (ARRAY['eci_winners_list'::text, 'gazette_notification'::text, 'party_confirmation'::text, 'physical_document'::text, 'other'::text])),
  evidence_note text,
  status text NOT NULL DEFAULT 'pending'::text CHECK (status = ANY (ARRAY['pending'::text, 'approved'::text, 'rejected'::text])),
  verified_by uuid,
  created_users_id uuid,
  created_at timestamp with time zone NOT NULL DEFAULT now(),
  decided_at timestamp with time zone,
  requesting_auth_id uuid,
  CONSTRAINT staff_verification_requests_pkey PRIMARY KEY (id),
  CONSTRAINT staff_verification_requests_app_id_fkey FOREIGN KEY (app_id) REFERENCES public.apps(id),
  CONSTRAINT staff_verification_requests_claimed_constituency_id_fkey FOREIGN KEY (claimed_constituency_id) REFERENCES public.constituencies(id),
  CONSTRAINT staff_verification_requests_verified_by_fkey FOREIGN KEY (verified_by) REFERENCES public.users(id),
  CONSTRAINT staff_verification_requests_created_users_id_fkey FOREIGN KEY (created_users_id) REFERENCES public.users(id),
  CONSTRAINT staff_verification_requests_requesting_auth_id_fkey FOREIGN KEY (requesting_auth_id) REFERENCES auth.users(id)
);
